Язык, который запускается за 15 минут

Один codex.exe, ноль зависимостей. Переменные, игры, файлы, JSON, HTTP-сервер — без фреймворков.

Скачать codex.exe Быстрый старт 
codex.exe new hello.cx
codex.exe hello.cx

Старт за 3 команды

codex.exe new hello.cx   # шаблон новичка
codex.exe hello.cx       # запуск
codex.exe test tests     # тесты с assert()

Примеры

secret := randint(1, 100)
tries := 0
while true {
    guess := num(input("Твой вариант: "))
    tries += 1
    if guess == secret { print("Точно за ", tries); break }
    else if guess < secret { print("Бери больше") }
    else { print("Бери меньше") }
}
fn hello(req) {
    if req["path"] == "/hi" {
        return to_json({"hi": "there", "time": date()})
    }
    return {"status": 404, "body": "try /hi"}
}
http_listen(8081, hello)
scores := [5, 3, 5, 2, 3]
print(sum(scores), avg(scores), unique(scores))
print(reduce(scores, add, 0))
print(find(scores, big), any(scores, big), all(scores, big))
print(chr(65), ord("A"), cwd(), clock())

Уроки и редактор

  • Туториалы docs/tutorial-01…11: basics → data → strings → files → functions → loops → maps → errors → packages → capstone → backend.
  • VS Code: папка editors/vscode — подсветка, сниппеты, codex fmt, fmt --check для CI.
  • Пакеты с GitHub без реестра: import "user/repo@ver", кэш ~/.codex/pkgs.
codex.exe help              # справка
codex.exe fmt --check a.cx  # проверка формата
codex.exe get user/repo     # скачать пакет
