Первая игра — через 15 минут, бэкенд — через 10 строк

CodeX — маленький язык, который читается как псевдокод, а работает как настоящий: переменные, массивы, структуры, файлы, JSON, HTTP-сервер, параллельность. Один codex.exe — и можно писать игры, ботов, API и скрипты для учебы.

guess.cx
secret := randint(1, 100)
tries := 0
print("Я загадал 1..100. Справишься?")
while true {
    guess := num(input("Твой вариант: "))
    tries += 1
    if guess == secret {
        print("Точно! Это " + str(secret) + " за " + str(tries))
        break
    } else if guess < secret {
        print("Бери больше")
    } else {
        print("Бери меньше")
    }
}
fn hello(req) {
    if req["path"] == "/hi" {
        return to_json({"hi": "there", "time": date()})
    }
    return {"status": 404, "body": "try /hi"}
}
print("API on 127.0.0.1:8081/hi")
http_listen(8081, hello)
