Examples

secret := randint(1, 100)
tries := 0
while true {
    guess := num(input("Your guess: "))
    tries += 1
    if guess == secret { print("Nailed it in ", tries); break }
    else if guess < secret { print("Go higher") }
    else { print("Go lower") }
}
fn hello(req) {
    if req["path"] == "/hi" {
        return to_json({"hi": "there", "time": date()})
    }
    return {"status": 404, "body": "try /hi"}
}
http_listen(8081, hello)
